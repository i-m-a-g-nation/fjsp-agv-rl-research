"""Bounded max-thinking diagnostic of existing source, no research approvals."""
from dataclasses import replace
from datetime import datetime, timedelta, timezone
import hashlib
import json
from pathlib import Path
import re
import subprocess
import sys
import time
from uuid import uuid4

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
import httpx
import langchain_openai
from src.ai_workflow.config import LLMConfig, load_dotenv_if_present
from src.ai_workflow.extraction_quality import extract_with_feedback
from src.ai_workflow.langgraph_pipeline import _llm_extract_paper
from src.ai_workflow.runtime_budget import BudgetLimits, RunBudget
from scripts.check_model_config import check_configuration


def main():
    load_dotenv_if_present()
    config = LLMConfig()
    public, errors = check_configuration(config)
    if errors or config.base_url != 'https://api.deepseek.com' or config.thinking_mode != 'enabled' or config.reasoning_effort != 'max':
        print('LOCAL_MAX_PROFILE_INVALID; no request sent')
        return 1
    now = datetime.now(timezone(timedelta(hours=8)))
    run = ROOT / 'experiments/runs' / ('model-api-max-' + now.strftime('%Y%m%d-%H%M%S') + '-' + uuid4().hex[:8])
    run.mkdir(parents=True, exist_ok=False)
    previous = json.loads((ROOT / 'experiments/results/model_api_validation_20261026_v1/result.json').read_bytes())
    limits = BudgetLimits(max_active_seconds=180, max_provider_calls=3, max_output_tokens=24576)
    budget = RunBudget(run / 'budget.sqlite', limits)
    cfg = replace(config, max_output_tokens=8192, request_timeout_seconds=60.0, max_retries=0, budget=budget)
    source_paths = ['src/ai_workflow/config.py', 'src/ai_workflow/extraction_quality.py',
                    'src/ai_workflow/langgraph_pipeline.py', 'src/ai_workflow/runtime_budget.py']
    record = {'kind': 'synthetic_max_api_diagnostic_not_research_acceptance', 'at': now.isoformat(),
        'base_commit': subprocess.check_output(['git','rev-parse','HEAD'], cwd=ROOT, text=True).strip(),
        'runtime_source_sha256': {p: hashlib.sha256((ROOT / p).read_bytes()).hexdigest() for p in source_paths},
        'configuration': public,
        'limits': {'calls': 3, 'output_tokens_reserved_max': 24576, 'request_timeout_seconds': 60, 'sdk_retries': 0},
        'http': [], 'probes': [], 'research_source_changed_by_run': False,
        'workflow_session_created': False, 'human_approval_created': False}

    def save():
        content = json.dumps(record, ensure_ascii=False, indent=2).replace(config.api_key, '<redacted>')
        (run / 'result.json').write_text(content+'\n', encoding='utf-8')

    def request_hook(request):
        body = json.loads(request.content)
        fields = {k: body[k] for k in ('model','thinking','reasoning_effort','max_tokens','max_completion_tokens') if k in body}
        assert fields.get('model') == 'deepseek-flash' and fields.get('thinking') == {'type':'enabled'}
        assert fields.get('reasoning_effort') == 'max' and fields.get('max_tokens') == 8192
        assert 'max_completion_tokens' not in fields
        record['http'].append({'path':request.url.path, 'request_fields':fields})
        save()

    def response_hook(response):
        response.read()
        item = record['http'][-1]
        item['http_status'] = response.status_code
        if response.status_code == 200:
            body = response.json()
            choice = body.get('choices',[{}])[0]
            message = choice.get('message',{})
            reasoning = message.get('reasoning_content') or ''
            usage = body.get('usage',{})
            item.update(finish_reason=choice.get('finish_reason'), reasoning_present=bool(reasoning),
                reasoning_characters=len(reasoning), usage={k:usage[k] for k in
                    ('prompt_tokens','completion_tokens','total_tokens','completion_tokens_details') if k in usage})
        # Do not store hidden reasoning, auth headers, or provider error bodies.
        save()

    original_client = langchain_openai.ChatOpenAI
    started = time.monotonic()
    budget.start_segment()
    probe = {'name':'known_tardiness_example', 'input':previous['probes'][0]['input']}
    record['probes'].append(probe)
    try:
        with httpx.Client(event_hooks={'request':[request_hook], 'response':[response_hook]}) as http_client:
            langchain_openai.ChatOpenAI = lambda **kw: original_client(http_client=http_client, **kw)
            client = langchain_openai.ChatOpenAI(api_key=cfg.api_key, base_url=cfg.base_url, model=cfg.model_name,
                temperature=cfg.temperature, timeout=cfg.provider_timeout(output_tokens=8192), max_retries=0,
                **cfg.chat_request_options())
            response = client.invoke([('system','Follow the supplied mathematical data and return only JSON.'),
                                      ('human',probe['input'])])
            probe['response'] = str(response.content)
            expected = previous['probes'][0]['expected']
            try:
                parsed = json.loads(str(response.content))
                probe['checks'] = {'strict_json_and_known_values': parsed == expected and all(type(v) is int for v in parsed.values()),
                    'not_truncated': record['http'][-1].get('finish_reason') == 'stop'}
            except (TypeError, ValueError):
                probe['checks'] = {'strict_json_and_known_values':False}
            probe['expected'] = expected
            save()

            fixture = previous['probes'][1]['input']
            probe = {'name':'extraction_with_bounded_feedback', 'input':fixture}
            record['probes'].append(probe)
            def once(feedback):
                return _llm_extract_paper(cfg,
                    'Synthetic API validation fixture (NOT a published paper)', fixture, 'full_text',
                    ['Manual synthetic fixture; no paper fetched, no RAG or codebase context.'],
                    'Only validate the existing extraction interface on synthetic input. Keep every absent RL field empty; '
                    'do not infer an algorithm, a reward, a network, or reusable source modules.', [], [], [], feedback=feedback)
            def attempt_snapshot(quality):
                probe['quality'] = quality
                save()
            extraction, quality = extract_with_feedback(cfg, once, paper_text=fixture, on_attempt=attempt_snapshot)
            objectives = re.sub(r'\s+|\\', '', ' '.join(extraction.math_model.objectives))
            def missing(value):
                if isinstance(value,list):
                    return all(missing(v) for v in value)
                return not value.strip() or any(t in value.lower() for t in
                    ('缺失','未提供','未给出','未说明','未明确','未指定','not specified','not provided'))
            probe['extraction'] = extraction.model_dump()
            probe['checks'] = {'all_rl_fields_missing': all(missing(v) for v in extraction.rl_formulation.model_dump().values()),
                'objective_formula_present': 'max(0,C_j-d_j)' in objectives and ('sum_j' in objectives or 'sum_{j}' in objectives),
                'citation_locations_complete': quality['status'] == 'complete',
                'still_unverified': extraction.evidence_status == 'unverified'}
    except Exception as exc:
        probe['error'] = {'type':type(exc).__name__, 'http_status':getattr(exc,'status_code',None)}
        if getattr(exc,'extraction_quality',None):
            probe['quality'] = exc.extraction_quality
    finally:
        langchain_openai.ChatOpenAI = original_client
        elapsed = time.monotonic()-started
        budget.record_active_seconds(elapsed)
        record['elapsed_seconds'] = round(elapsed,3)
        record['budget'] = budget.snapshot()
        record['all_requests_max'] = bool(record['http']) and all(
            r['request_fields'].get('thinking') == {'type':'enabled'} and r['request_fields'].get('reasoning_effort') == 'max'
            for r in record['http'])
        record['passed'] = len(record['probes']) == 2 and all(p.get('checks') and all(p['checks'].values()) for p in record['probes']) and all(
            r.get('http_status') == 200 and r.get('finish_reason') == 'stop' for r in record['http'])
        save()
    print(json.dumps({'passed':record['passed'], 'all_requests_max':record['all_requests_max'],
        'elapsed_seconds':record['elapsed_seconds'], 'provider_calls_reserved':record['budget']['provider_calls'],
        'probes':[{k:p[k] for k in ('name','checks','error') if k in p} for p in record['probes']],
        'http':record['http'], 'artifact':str(run/'result.json')},ensure_ascii=False))
    return 0 if record['passed'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
