"""One-off diagnostic of existing interfaces; no workflow approval or research run."""
from dataclasses import replace
from datetime import datetime, timedelta, timezone
import json
from pathlib import Path
import subprocess
import sys
import time
from uuid import uuid4

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from langchain_openai import ChatOpenAI
from scripts.check_model_config import check_configuration
from src.ai_workflow.config import LLMConfig, load_dotenv_if_present
from src.ai_workflow.langgraph_pipeline import _llm_extract_paper
from src.ai_workflow.runtime_budget import BudgetLimits, RunBudget


def main():
    load_dotenv_if_present()
    config = LLMConfig()
    public, errors = check_configuration(config)
    if errors:
        print('LOCAL_CONFIG_INVALID; no provider request sent')
        return 1
    if config.base_url != 'https://api.deepseek.com' or config.thinking_mode != 'disabled':
        print('DIAGNOSTIC_EXPECTS_OFFICIAL_NONTHINKING_CONFIGURATION; no request sent')
        return 1
    now = datetime.now(timezone(timedelta(hours=8)))
    run = ROOT / 'experiments' / 'runs' / ('model-api-validation-' + now.strftime('%Y%m%d-%H%M%S') + '-' + uuid4().hex[:8])
    run.mkdir(parents=True, exist_ok=False)
    limits = BudgetLimits(max_active_seconds=90, max_provider_calls=2, max_output_tokens=2304)
    budget = RunBudget(run / 'budget.sqlite', limits)
    cfg = replace(config, request_timeout_seconds=30.0, max_retries=0, budget=budget)
    record = {
        'kind': 'synthetic_api_diagnostic_not_research_acceptance',
        'at': now.isoformat(),
        'source_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
        'configuration': public,
        'limits': {'calls': 2, 'output_tokens_reserved_max': 2304, 'per_request_timeout_seconds': 30, 'retries': 0},
        'probes': [],
        'workflow_session_created': False,
        'human_approval_created': False,
        'research_source_modified': False,
    }

    def save():
        # API key never appears in prompts, public configuration, output or exceptions.
        content = json.dumps(record, ensure_ascii=False, indent=2).replace(config.api_key, '<redacted>')
        (run / 'result.json').write_text(content + '\n', encoding='utf-8')

    def missing(value):
        if isinstance(value, list):
            return all(missing(item) for item in value)
        return not value.strip() or any(token in value.lower() for token in
            ('缺失', '未提供', '未给出', '未说明', '未明确', '未指定', 'not specified', 'not provided'))

    started = time.monotonic()
    budget.start_segment()
    probe = {'name': 'known_tardiness_example', 'max_output_tokens': 256, 'connected': False}
    record['probes'].append(probe)
    try:
        prompt = (
            'Synthetic API diagnostic, not a research claim. One machine processes A then B; both are released at 0. '
            'Processing times are 2 and 2; due dates are 2 and 4. One compulsory non-preemptive maintenance has '
            'duration 2 and its complete interval must lie in [0,6). Jobs and maintenance cannot overlap. '
            'Case fixed: maintenance starts at 0. Case joint: choose the maintenance start minimizing total tardiness. '
            'Tardiness is max(0, completion - due date). Return only a JSON object with integer fields '
            'fixed_total_tardiness, joint_total_tardiness, joint_maintenance_start. No explanation.'
        )
        probe['input'] = prompt
        first_cfg = replace(cfg, max_output_tokens=256)
        client = ChatOpenAI(api_key=first_cfg.api_key, base_url=first_cfg.base_url,
            model=first_cfg.model_name, temperature=first_cfg.temperature,
            timeout=first_cfg.provider_timeout(output_tokens=256), max_retries=0,
            **first_cfg.chat_request_options())
        then = time.monotonic()
        response = client.invoke([('system', 'Follow the supplied mathematical data and return only JSON.'), ('human', prompt)])
        probe.update(connected=True, elapsed_seconds=round(time.monotonic()-then, 3),
            response=str(response.content), metadata=response.response_metadata, usage=response.usage_metadata)
        expected = {'fixed_total_tardiness': 4, 'joint_total_tardiness': 0, 'joint_maintenance_start': 4}
        probe['expected'] = expected
        try:
            parsed = json.loads(str(response.content))
            probe['checks'] = {'strict_json_and_known_values': parsed == expected and all(type(v) is int for v in parsed.values()),
                'not_truncated': response.response_metadata.get('finish_reason') == 'stop'}
        except (TypeError, ValueError):
            probe['checks'] = {'strict_json_and_known_values': False}
        save()

        fixture = (
            'TEST FIXTURE: This is synthetic API validation data, not a published paper.\n'
            'We study scheduling of production jobs on identical parallel machines with compulsory maintenance.\n'
            'Every machine has exactly one non-preemptive maintenance, including idle machines.\n'
            'The complete maintenance interval must fit inside its allowed maintenance window.\n'
            'The objective is to minimize total tardiness, defined as sum_j max(0, C_j - d_j).\n'
            'C_j denotes completion time and d_j denotes the due date of job j.\n'
            'No reward function, state space, action space, action masking, or neural network architecture is specified.\n'
        )
        probe = {'name': 'existing_extraction_with_missing_rl_fields', 'max_output_tokens': 2048,
            'input': fixture, 'connected': False}
        record['probes'].append(probe)
        then = time.monotonic()
        extraction = _llm_extract_paper(replace(cfg, max_output_tokens=2048),
            'Synthetic API validation fixture (NOT a published paper)', fixture, 'full_text',
            ['Manual synthetic fixture; no paper fetched, no RAG or codebase context.'],
            'Only validate the existing extraction interface on synthetic input. Keep every absent RL field empty; '
            'do not infer an algorithm, a reward, a network, or reusable source modules.', [], [], [])
        values = extraction.model_dump()
        audit = extraction.evidence_audit['summary']
        objectives = ' '.join(extraction.math_model.objectives)
        probe.update(connected=True, elapsed_seconds=round(time.monotonic()-then, 3), extraction=values,
            checks={'all_rl_fields_missing': all(missing(v) for v in extraction.rl_formulation.model_dump().values()),
                'objective_present': bool(objectives) and ('延期' in objectives or 'tardiness' in objectives.lower()),
                'all_claim_quotes_located': audit['all_claims_located'],
                'still_unverified': extraction.evidence_status == 'unverified',
                'llm_extraction_not_offline_fallback': extraction.extraction_method == 'llm'})
    except Exception as exc:
        probe['error'] = {'type': type(exc).__name__, 'http_status': getattr(exc, 'status_code', None)}
        # No exception message/body: providers can echo user configuration.
    finally:
        elapsed = time.monotonic()-started
        budget.record_active_seconds(elapsed)
        record['elapsed_seconds'] = round(elapsed, 3)
        record['budget'] = budget.snapshot()
        record['passed'] = len(record['probes']) == 2 and all(
            p.get('connected') and p.get('checks') and all(p['checks'].values()) for p in record['probes'])
        save()
    print(json.dumps({'passed': record['passed'], 'elapsed_seconds': record['elapsed_seconds'],
        'provider_calls_reserved': record['budget']['provider_calls'],
        'output_tokens_reserved': record['budget']['output_tokens_reserved'],
        'probes': [{k: p[k] for k in ('name', 'connected', 'checks', 'error') if k in p} for p in record['probes']],
        'artifact': str(run / 'result.json')}, ensure_ascii=False))
    return 0 if record['passed'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
